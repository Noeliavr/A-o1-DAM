create table usuarios (
id_usuario int primary key,
nombre varchar(50),
activo boolean
);

create table prestamos (
id_prestamo int primary key,
id_usuario int,
fecha_prestamo date,
fecha_devolucion date,
foreign key (id_usuario) references usuarios(id_usuario)
);

Punto 1:

insert into usuarios_frecuentes
(id_usuario,
nombre,
total_prestamos)
select
us.id_usuario as id_usuario
us.nombre as nombre
count(id_prestamos) as total
from
usuarios us 
inner join prestamos pr
on us.id_usuario=pr.id_usuario
group by us.id_usuario
having total >
  (select
   avg (total.total) media
   from 
     (select 
      id_usuario,
      count(id_prestamo) as total
      from prestamos
      group by id_usuario) total);

set activo=0
where id_usuario in
  (select id_usuario
   from prestamos
   where fecha_devolucion is null);

delete from prestamos

create table usuarios_sin_prestamos as
select * 
from usuarios
where id_usuario not in
  (select id_usuario 
   from prestamos);

Punto 5:

delete from usuarios
where id_usuario in
  (select id_usuario 
   from usuarios_sin_prestamos);
