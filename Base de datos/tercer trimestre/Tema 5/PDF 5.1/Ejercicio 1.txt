create table clientes (
id_cliente int primary key,
nombre varchar(50),
ciudad varchar(50),
activo boolean
);

create table pedidos (
id_pedido int primary key,
id_cliente int,
total decimal(10,2),
fecha date,
foreign key (id_cliente) references clientes(id_cliente)
);

Respuestas:

Punto 1:

create table clientes_premium (
id_cliente int,
nombre varchar(50),
total_compras decimal(10,2)
);

insert into clientes_premium 
(id_cliente,
 nombre,
 total_compras)
select 
cl.id_cliente as id_cliente,
cl.nombre as nombre,
sum(pe.total) as total
from clientes cl
inner join pedidos pe
on cl.id_cliente=pe.id_cliente
group by cl.id_cliente
having total > 
  (select avg(total) 
   from pedidos);

Punto 2:

update clientes
set activo=FALSE
where id_cliente in
(select pe.id_cliente as id_cliente
from pedidos pe
group by pe.id_cliente
having sum(pe.total)<(select avg(total)from pedidos));
